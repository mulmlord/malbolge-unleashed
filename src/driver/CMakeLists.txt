configure_file(version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp" @ONLY)

# The RISC-V runtime as byte arrays (runtime_files.hpp), for `malm build` of C, C++ and Rust sources.
set(rt_dir "${PROJECT_SOURCE_DIR}/rv/runtime")
file(GLOB rt_files CONFIGURE_DEPENDS "${rt_dir}/include/*.h" "${rt_dir}/src/*.c"
    "${rt_dir}/rust/*.rs")
set(rt_embed "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime_files.cpp")
add_custom_command(OUTPUT "${rt_embed}"
    COMMAND "${CMAKE_COMMAND}" -DRUNTIME_DIR=${rt_dir} -DOUTPUT=${rt_embed}
        -P "${PROJECT_SOURCE_DIR}/cmake/EmbedRuntime.cmake"
    DEPENDS "${PROJECT_SOURCE_DIR}/cmake/EmbedRuntime.cmake" "${rt_dir}/crt0.S" "${rt_dir}/link.ld"
        ${rt_files}
    COMMENT "Embedding the RISC-V runtime"
    VERBATIM)

# The picolibc sysroot archive (rv/runtime/CMakeLists.txt), included by the assembler, or none.
set(pl_embed "${CMAKE_CURRENT_BINARY_DIR}/generated/picolibc_sysroot.cpp")
if(TARGET picolibc_sysroot)
    get_target_property(pl_archive picolibc_sysroot MALM_ARCHIVE)
    file(CONFIGURE OUTPUT "${pl_embed}" @ONLY CONTENT [=[
// Generated by src/driver/CMakeLists.txt; do not edit.
#include "driver/runtime_files.hpp"

__asm__(".section .rodata\n"
        ".balign 16\n"
        "malm_picolibc_begin:\n"
        ".incbin \"@pl_archive@\"\n"
        "malm_picolibc_end:\n"
        ".previous\n");
extern "C" const char malm_picolibc_begin[];
extern "C" const char malm_picolibc_end[];

std::string_view malm::picolibc_sysroot() {
    return {malm_picolibc_begin, static_cast<std::size_t>(malm_picolibc_end - malm_picolibc_begin)};
}
]=])
    set_source_files_properties("${pl_embed}" PROPERTIES OBJECT_DEPENDS "${pl_archive}")
else()
    file(CONFIGURE OUTPUT "${pl_embed}" CONTENT [=[
// Generated by src/driver/CMakeLists.txt; do not edit. malm is built without picolibc.
#include "driver/runtime_files.hpp"

std::string_view malm::picolibc_sysroot() { return {}; }
]=])
endif()

add_library(malm_driver asm_cmd.cpp build_cmd.cpp hla_cmd.cpp cli.cpp disasm_cmd.cpp run_cmd.cpp debug_observer.cpp export_cmd.cpp pack_cmd.cpp packed_run.cpp program_io.cpp rvref_cmd.cpp rvemu_cmd.cpp rvaot_cmd.cpp sysroot.cpp toolchain.cpp "${rt_embed}" "${pl_embed}")
target_include_directories(malm_driver
    PUBLIC "${PROJECT_SOURCE_DIR}/src"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(malm_driver PUBLIC malm_core malm_asm malm_hla malm_container malm_pack malm_rv malm_aot PRIVATE malm_warnings libzstd_static)
if(TARGET picolibc_sysroot)
    add_dependencies(malm_driver picolibc_sysroot)
endif()

if(MALM_STATIC)
    target_compile_definitions(malm_driver PUBLIC MALM_STATIC)
endif()

add_executable(malm main.cpp)
target_link_libraries(malm PRIVATE malm_driver malm_warnings)
if(MALM_STATIC)
    target_link_options(malm PRIVATE -static)
endif()
